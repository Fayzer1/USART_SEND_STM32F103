# USART_SEND_STM32F103
Подпрограмма, реализующая отправку данных в асинхронном режиме через модуль USART2. В регистре R1 хранится указатель на начало отправляемой строки, признаком конца строки являются два подряд идущих непечатных символа 0x0D и 0x0A.

```
/**
 ******************************************************************************
 * @file      startup_stm32f103rbtx.s
 * @author    Auto-generated by STM32CubeIDE
 * @brief     STM32F103RBTx device vector table for GCC toolchain.
 *            This module performs:
 *                - Set the initial SP
 *                - Set the initial PC == Reset_Handler,
 *                - Set the vector table entries with the exceptions ISR address
 *                - Branches to main in the C library (which eventually
 *                  calls main()).
 ******************************************************************************
 * @attention
 *
 * Copyright (c) 2026 STMicroelectronics.
 * All rights reserved.
 *
 * This software is licensed under terms that can be found in the LICENSE file
 * in the root directory of this software component.
 * If no LICENSE file comes with this software, it is provided AS-IS.
 *
 ******************************************************************************
 */

.syntax unified
.cpu cortex-m3
.fpu softvfp
.thumb
.include "stm32f10x.s"

.global g_pfnVectors
.global Default_Handler

/* start address for the initialization values of the .data section.
defined in linker script */
.word _sidata
/* start address for the .data section. defined in linker script */
.word _sdata
/* end address for the .data section. defined in linker script */
.word _edata
/* start address for the .bss section. defined in linker script */
.word _sbss
/* end address for the .bss section. defined in linker script */
.word _ebss

/**
 * @brief  This is the code that gets called when the processor first
 *          starts execution following a reset event. Only the absolutely
 *          necessary set is performed, after which the application
 *          supplied main() routine is called.
 * @param  None
 * @retval : None
*/

  .section .text.Reset_Handler
  .weak Reset_Handler
  .type Reset_Handler, %function
Reset_Handler:
 	ldr   r0, =_estack
 	mov   sp, r0          /* set stack pointer */

										@ включаем тактирование
	ldr		r0, =RCC_APB2ENR			@ портов A,C и AFIO
	mov		r1, #0x15
	ldr		r2, [r0]
	bfi		r2, r1, #0, #5
	str		r2, [r0]

 										@ включаем тактирование usart2
	ldr		r0, =PERIPH_BB_BASE+ \
			(RCC_APB1ENR-PERIPH_BASE)*32 + \
			17*4

	mov		r1, #1
	str 	r1, [r0]

										@ настройка порта Tx
	ldr		r0, =GPIOA_CRL
	mov		r1, #0x9					@ 4-битная маска настроек для
	ldr		r2, [r0]					@ Output mode 10MHz,
    bfi		r2, r1, #8, #4				@ Alternate Function Push-Pull ("1001")
    str		r2, [r0]

										@ настройка скорости
    ldr		r0, =USART2_BRR

	mov		r1, #0x8D					@ маска для настройки скорости передачи
	ldr		r2, [r0]					@ 57600, 8МГц
    bfi		r2, r1, #0, #16
    str		r2, [r0]

										@ настройка uart
    ldr		r0, =USART2_CR1
	mov		r1, #0x2008					@ маска настроек для включения UE и TE
	ldr		r2, [r0]
    bfi		r2, r1, #0, #14
    str		r2, [r0]

										@ проверка нажатия кнопки
button:
	ldr		r0, =GPIOC_IDR
	ldr		r1, [r0]
	tst		r1,	#GPIO_IDR_IDR13
	it		EQ
	bleq	delay
	tst		r1, #GPIO_IDR_IDR13
	beq		send
	bne		button
										@ защита от дребезга
delay:
	ldr		r3, =0xFFFFF				@1048576/8000000 ~ 130мс
delay_loop:
	subs	r3, #1
	bne		delay_loop
	bx 		lr


send:									@ подпрограмма отправки по usart2
	push	{r0,r1,r2,r3,r4}
	ldr		r1, =content				@ записываем указатель на строку в R1
	ldr		r2,	=USART2_DR
	ldr		r3, =USART2_SR

send_byte:
	mov		r5, #0
	bfi		r4, r5, #6, #1				@ сброс TC
	str		r4,	[r3]
	ldrb	r0, [r1], #1
	str		r0, [r2]
	bl		transmitt					@ проверка TC
	cmp		r0, #13						@ проверка конца строки
	beq		str_end
	b		send_byte

transmitt:								@ блок проверки флага TC
	ldr		r4, [r3]
	tst		r4,	#USART_SR_TC
	beq		transmitt
	bx 		lr

str_end:								@ блок проверки конца строки
	ldrb	r0, [r1]
	cmp		r0, #10
	bne		send_byte
	str		r0,	[r2]
	bl		transmitt
	pop		{r0,r1,r2,r3,r4}
	b		button


content:
	.ascii "Knock! Knock!\r\n" 			@ запись строки в код программы
										@ добавление знаков конца строки

  	.size Reset_Handler, .-Reset_Handler

/**
 * @brief  This is the code that gets called when the processor receives an
 *         unexpected interrupt.  This simply enters an infinite loop, preserving
 *         the system state for examination by a debugger.
 *
 * @param  None
 * @retval : None
*/
  .section .text.Default_Handler,"ax",%progbits
Default_Handler:
Infinite_Loop:
  b Infinite_Loop
  .size Default_Handler, .-Default_Handler
```
